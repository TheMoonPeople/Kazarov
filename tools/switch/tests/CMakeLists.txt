cmake_minimum_required(VERSION 3.25)
project(kazarov_switch_archive_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(KZ_TEST_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(ZLIB REQUIRED)
find_library(KZ_TEST_ZSTD NAMES zstd REQUIRED)
find_path(KZ_TEST_ZSTD_INCLUDE_DIR zstd.h REQUIRED)
add_executable(kazarov_archive_probe archive_probe.cpp "${CMAKE_CURRENT_LIST_DIR}/../src/zip_archive.cpp")
target_include_directories(kazarov_archive_probe PRIVATE "${KZ_TEST_ROOT}/include" "${KZ_TEST_ZSTD_INCLUDE_DIR}")
target_link_libraries(kazarov_archive_probe PRIVATE ZLIB::ZLIB "${KZ_TEST_ZSTD}")
enable_testing()
find_package(Threads REQUIRED)
add_executable(kazarov_exit_registry_test exit_registry_test.cpp "${CMAKE_CURRENT_LIST_DIR}/../src/exit.c")
target_include_directories(kazarov_exit_registry_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/newlib_abi")
target_compile_definitions(kazarov_exit_registry_test PRIVATE _GNU_SOURCE)
target_compile_options(kazarov_exit_registry_test PRIVATE -UNDEBUG)
target_link_libraries(kazarov_exit_registry_test PRIVATE Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(kazarov_exit_registry_test PRIVATE KZ_TEST_WRAP_CALLOC)
    target_link_options(kazarov_exit_registry_test PRIVATE -Wl,--wrap=calloc)
endif()
add_test(NAME native_exit_registry COMMAND kazarov_exit_registry_test)
set_tests_properties(native_exit_registry PROPERTIES TIMEOUT 15)
add_test(NAME native_zip_regression COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/archive_regression.py"
    --probe "$<TARGET_FILE:kazarov_archive_probe>")
